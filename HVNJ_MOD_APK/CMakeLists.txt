cmake_minimum_required(VERSION 3.22)
project(HVNJMod LANGUAGES C CXX ASM)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "This profile supports Android ARM64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(DynamicBinaryInstrument OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/dobby-2021/Dobby-31e4ab0cfc2bfa4512b4ea520786ea2403f97dfc dobby2021 EXCLUDE_FROM_ALL)
# Old logging.c includes android/log.h inside a function. NDK29 adds inline
# definitions, so preinclude it at file scope without modifying pinned sources.
target_compile_options(dobby PRIVATE -include android/log.h)
add_library(hvnjmod SHARED src/hv_runtime.cpp src/hv_hooks.cpp src/hv_menu.cpp src/hv_features.cpp src/hv_bag.cpp)
target_include_directories(hvnjmod PRIVATE build/generated third_party/dobby-2021/Dobby-31e4ab0cfc2bfa4512b4ea520786ea2403f97dfc/include)
target_compile_options(hvnjmod PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
target_link_options(hvnjmod PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
target_link_libraries(hvnjmod PRIVATE dobby log dl)
